/*
 * lcc back end for the Hitachi H8/500 (H8/532 in maximum mode), output
 * for Alfred Arnold's AS (asl).  Small model: all code in page 0 (JSR /
 * RTS, 16-bit code addresses), all data in page 8 (DP = EP = TP = 8,
 * 16-bit data pointers); constants and initialized data are copied from
 * ROM to RAM by the start-up code (crt0.asm).
 *
 * Types: char 8, short/int/pointer 16, long 32, float/double 32 (helper
 * calls).  Registers: r0-r5 allocatable (r0-r3 caller-saved, r4-r5
 * callee-saved), r6 = FP (LINK/UNLK), r7 = SP.  32-bit values live in
 * the even pairs r0:r1, r2:r3, r4:r5 (high word in the even register,
 * as in memory: big-endian); their code is generated in emit2().
 * Arguments are pushed right to left, 2-byte aligned, and popped by
 * the caller; results in r0 or r0:r1.
 *
 * Part of moppe (tools/h8500); the rest of lcc is by Fraser and Hanson,
 * see lcc's CPYRIGHT.
 */

static int h8sametree(Node p, Node q) {
	return p == NULL && q == NULL
	|| p && q && p->op == q->op && p->syms[0] == q->syms[0]
		&& h8sametree(p->kids[0], q->kids[0])
		&& h8sametree(p->kids[1], q->kids[1]);
}
/* ASGN(a, OP(INDIR(a), ..)): a read-modify-write of one location */
static int h8memop(Node p) {
	if (generic(p->kids[1]->kids[0]->op) == INDIR
	&& h8sametree(p->kids[0], p->kids[1]->kids[0]->kids[0]))
		return 3;
	return LBURG_MAX;
}
/* a 16-bit LOAD is a register copy (lcc may rename it away) unless it
   narrows a 32-bit pair: that one takes the low word */
static int move2(Node p) {
	return opsize(p->kids[0]->op) == 4 ? 1 : move(p);
}

#define R(n) (ireg[n]->x.name)

static void progbeg(int argc, char *argv[]) {
	int i;

	parseflags(argc, argv);
	for (i = 1; i < argc; i++)
		if (strncmp(argv[i], "-tag=", 5) == 0)
			tag = argv[i] + 5;
	for (i = 0; i < 6; i++)
		ireg[i] = mkreg("r%d", i, 1, IREG);
	for (i = 0; i < 6; i += 2)
		preg[i] = mkreg("r%d", i, 3, IREG);
	iregw = mkwildcard(ireg);
	pregw = mkwildcard(preg);
	tmask[IREG] = 0x0F;		/* r0-r3 */
	vmask[IREG] = 0x30;		/* r4-r5 */
	tmask[FREG] = vmask[FREG] = 0;
	quo = mkreg("r1", 1, 1, IREG);
	quo->x.regnode->mask |= 1;
	rem = mkreg("r0", 0, 1, IREG);
	rem->x.regnode->mask |= 2;
	cseg = -1;
	print("; lcc h8500 tag %s\n", tag);
}

static Symbol rmap(int opk) {
	switch (optype(opk)) {
	case B: case P:
		return iregw;
	case I: case U: case F:
		return opsize(opk) == 4 ? pregw : iregw;
	default:
		return 0;
	}
}

static void progend(void) {
	segment(0);
}

static void target(Node p) {
	assert(p);
	switch (specific(p->op)) {
	case CALL+I: case CALL+U: case CALL+P: case CALL+F:
		setreg(p, opsize(p->op) == 4 ? preg[0] : ireg[0]);
		break;
	/* RET is not targeted: a two-address kid pinned to r0 could clash
	   with a right operand fixed in r0 (a%b in r0); emit2 moves it */
	case MUL+I: case MUL+U:
		if (opsize(p->op) == 2) {
			setreg(p, quo);			/* low word in r1, r0 clobbered */
			rtarget(p, 0, ireg[0]);
			break;
		}
		/* fall through: 32-bit helpers */
	case DIV+I: case DIV+U: case MOD+I: case MOD+U:
		if (opsize(p->op) == 2) {
			setreg(p, generic(p->op) == DIV ? quo : rem);
			rtarget(p, 0, ireg[1]);
			rtarget(p, 1, ireg[2]);
			break;
		}
		setreg(p, preg[0]);
		rtarget(p, 0, preg[0]);
		rtarget(p, 1, preg[2]);
		break;
	case LSH+I: case LSH+U: case RSH+I: case RSH+U:
		if (generic(cse(p->kids[1])->op) == CNST)
			break;
		if (opsize(p->op) == 2) {
			setreg(p, ireg[1]);
			rtarget(p, 0, ireg[1]);
			rtarget(p, 1, ireg[2]);
		} else {
			setreg(p, preg[0]);
			rtarget(p, 0, preg[0]);
			rtarget(p, 1, ireg[2]);
		}
		break;
	case ADD+F: case SUB+F: case MUL+F: case DIV+F:
	case EQ+F: case NE+F: case GE+F: case GT+F: case LE+F: case LT+F:
		if (generic(p->op) != EQ && generic(p->op) != NE && generic(p->op) != GE
		&& generic(p->op) != GT && generic(p->op) != LE && generic(p->op) != LT)
			setreg(p, preg[0]);
		rtarget(p, 0, preg[0]);
		rtarget(p, 1, preg[2]);
		break;
	case CVF+I:
		setreg(p, opsize(p->op) == 4 ? preg[0] : ireg[1]);
		rtarget(p, 0, preg[0]);
		break;
	case CVI+F:
		setreg(p, preg[0]);
		rtarget(p, 0, opsize(p->kids[0]->op) == 4 ? preg[0] : ireg[1]);
		break;
	case ASGN+B:
		rtarget(p, 0, ireg[2]);
		rtarget(p->kids[1], 0, ireg[3]);
		break;
	case ARG+B:
		rtarget(p->kids[0], 0, ireg[3]);
		break;
	}
}

static void clobber(Node p) {
	assert(p);
	switch (specific(p->op)) {
	case CALL+I: case CALL+U: case CALL+P:
		spill(opsize(p->op) == 4 ? 0x0C : 0x0E, IREG, p);
		break;
	case CALL+F:
		spill(0x0C, IREG, p);
		break;
	case CALL+V:
		spill(0x0F, IREG, p);
		break;
	case ASGN+B: case ARG+B:
		spill(0x03, IREG, p);
		break;
	case DIV+I: case MOD+I:
		if (opsize(p->op) == 4)
			break;
		break;
	case EQ+F: case NE+F: case GE+F: case GT+F: case LE+F: case LT+F:
		spill(0x01, IREG, p);	/* __cmpf returns -1/0/1 in r0 */
		break;
	case CVF+I:
		if (opsize(p->op) == 2)
			spill(0x01, IREG, p);
		break;
	}
}

static int rnum(Node p) {
	/* a register variable read has its register on the VREG symbol */
	if (generic(p->op) == INDIR && p->kids[0] && specific(p->kids[0]->op) == VREG+P
	&& !p->syms[RX])
		return p->kids[0]->syms[0]->x.regnode->number;
	return getregnum(p);
}

/* a read of a temporary that holds a constant or an address: lcc's
   reuse() (gen.c) reduces it as that tree when the rule matches it at no
   cost (cnst, addr4), so look at what will actually be emitted */
static Node cse(Node p) {
	Symbol r = p->syms[RX];
	if (generic(p->op) == INDIR && p->kids[0] && p->kids[0]->op == VREG+P
	&& r && r->temporary && r->u.t.cse) {
		int op = generic(r->u.t.cse->op);
		if (op == CNST || op == ADDRG || op == ADDRL || op == ADDRF)
			return r->u.t.cse;
	}
	return p;
}

static int isaddrop4(Node a) {
	return generic(a->op) == ADDRG || generic(a->op) == ADDRL || generic(a->op) == ADDRF;
}

/* the effective address of a 32-bit object (high word at +0, low at +2):
   a is the address tree, areg the node holding it when in a register */
static char *ea4(Node a, Node areg, int half) {
	switch (generic(a->op)) {
	case ADDRG:
		return stringf("@%s+%d", a->syms[0]->x.name, half);
	case ADDRL: case ADDRF:
		return stringf("@(%d,r6)", a->syms[0]->x.offset + half);
	default:
		return half ? stringf("@(2,r%d)", rnum(areg)) : stringf("@r%d", rnum(areg));
	}
}

static void mov4(int dst, int src) {
	if (dst != src)
		print("mov.w r%d,r%d\nmov.w r%d,r%d\n", src, dst, src + 1, dst + 1);
}

static void shift2(int r, int op, int n) {
	char *ins = op == LSH ? "shll.w" : op == RSH + U ? "shlr.w" : "shar.w";
	n &= 15;
	if (n >= 8 && op != RSH + I) {		/* whole bytes with SWAP */
		if (op == LSH)
			print("swap.b r%d\nand.w #$FF00,r%d\n", r, r);
		else
			print("swap.b r%d\nand.w #$00FF,r%d\n", r, r);
		n -= 8;
	}
	while (n-- > 0)
		print("%s r%d\n", ins, r);
}

static void shift4(int r, int op, int n) {
	n &= 31;
	if (n >= 16) {
		if (op == LSH)
			print("mov.w r%d,r%d\nclr.w r%d\n", r + 1, r, r + 1);
		else if (op == RSH + U)
			print("mov.w r%d,r%d\nclr.w r%d\n", r, r + 1, r);
		else {
			int l = genlabel(1);
			print("mov.w r%d,r%d\nclr.w r%d\ntst.w r%d\nbpl L_%s_s%d\nmov.w #-1,r%d\nL_%s_s%d:\n",
				r, r + 1, r, r + 1, tag, l, r, tag, l);
		}
		n -= 16;
		if (op == LSH)
			shift2(r, LSH, n);
		else
			shift2(r + 1, op, n);
		return;
	}
	while (n-- > 0) {
		if (op == LSH)
			print("shll.w r%d\nrotxl.w r%d\n", r + 1, r);
		else
			print("%s r%d\nrotxr.w r%d\n", op == RSH + U ? "shlr.w" : "shar.w", r, r + 1);
	}
}

static char *bcc(int op, int high) {
	/* branch for a 32-bit comparison: on the high words (signed or not),
	   and the unsigned one used for the low words; floats branch on
	   __cmpf's signed -1/0/1 */
	if (optype(op) == F)
		op = generic(op) + I;
	switch (generic(op)) {
	case EQ: return "beq";
	case NE: return "bne";
	case GE: return high && optype(op) == I ? "bge" : "bcc";
	case GT: return high && optype(op) == I ? "bgt" : "bhi";
	case LE: return high && optype(op) == I ? "ble" : "bls";
	case LT: return high && optype(op) == I ? "blt" : "bcs";
	}
	assert(0);
	return "";
}

static void emit2(Node p) {
	int op = specific(p->op), sz = opsize(p->op);
	int dst, src;

	if ((generic(op) == INDIR || generic(op) == ASGN) && p->kids[0]
	&& specific(p->kids[0]->op) == VREG+P)
		return;			/* register variable read / write */
	if (generic(op) == RET) {
		src = rnum(p->x.kids[0]);
		if (sz == 4)
			mov4(0, src);
		else if (src != 0)
			print("mov.w r%d,r0\n", src);
		return;
	}

	switch (generic(op)) {
	case LOAD: case CVF:
		if (generic(op) == CVF && optype(op) != F)
			break;
		dst = rnum(p);
		src = rnum(p->x.kids[0]);
		if (sz == 4)
			mov4(dst, src);
		else {
			if (opsize(p->x.kids[0]->op) == 4)
				src++;		/* a narrowing LOAD: the low word */
			if (dst != src)
				print("mov.w r%d,r%d\n", src, dst);
		}
		return;
	case CVI: case CVU: case CVP: {
		int ssz = opsize(p->x.kids[0]->op);
		if (getenv("H8DEBUG"))
			print(";cv %s kid %s ssz %d\n", opname(p->op), opname(p->x.kids[0]->op), ssz);
		dst = rnum(p);
		src = rnum(p->x.kids[0]);
		if (optype(op) == F) {		/* CVIF4 */
			print(ssz == 4 ? "jsr @__itof4\n" : "jsr @__itof2\n");
			return;
		}
		if (sz <= 2 && ssz == 4) {		/* truncate: the low word */
			if (dst != src + 1)
				print("mov.w r%d,r%d\n", src + 1, dst);
		} else if (sz <= 2) {
			if (dst != src)
				print("mov.w r%d,r%d\n", src, dst);
			if (sz == 2 && ssz == 1)
				print(generic(op) == CVI ? "exts.b r%d\n" : "extu.b r%d\n", dst);
		} else if (ssz == 4) {
			mov4(dst, src);
		} else {				/* extend to 32 bits */
			if (dst + 1 != src)
				print("mov.w r%d,r%d\n", src, dst + 1);
			if (ssz == 1)
				print(generic(op) == CVI ? "exts.b r%d\n" : "extu.b r%d\n", dst + 1);
			if (generic(op) == CVI) {
				int l = genlabel(1);
				print("clr.w r%d\ntst.w r%d\nbpl L_%s_x%d\nmov.w #-1,r%d\nL_%s_x%d:\n",
					dst, dst + 1, tag, l, dst, tag, l);
			} else
				print("clr.w r%d\n", dst);
		}
		return;
	}
	case LSH: case RSH: {
		Node k = cse(p->kids[1]);
		dst = rnum(p);
		src = rnum(p->x.kids[0]);
		if (generic(k->op) == CNST) {
			int n = k->syms[0]->u.c.v.i;
			int kind = generic(op) == LSH ? LSH : optype(op) == U ? RSH + U : RSH + I;
			if (sz == 4) {
				mov4(dst, src);
				shift4(dst, kind, n);
			} else {
				if (dst != src)
					print("mov.w r%d,r%d\n", src, dst);
				shift2(dst, kind, n);
			}
		} else {				/* value in r1, count in r2 */
			int l = genlabel(2);
			char *ins = generic(op) == LSH ? "shll.w" : optype(op) == U ? "shlr.w" : "shar.w";
			print("L_%s_l%d:\ntst.w r2\nbeq L_%s_l%d\n%s r1\nadd.w #-1,r2\nbra L_%s_l%d\nL_%s_l%d:\n",
				tag, l, tag, l + 1, ins, tag, l, tag, l + 1);
		}
		return;
	}
	case CNST: {
		unsigned long v = sz == 4 && optype(op) == F ? 0 : p->syms[0]->u.c.v.u;
		dst = rnum(p);
		if (optype(op) == F) {
			float f = p->syms[0]->u.c.v.d;
			v = *(unsigned *)&f;
		}
		print("mov.w #%d,r%d\nmov.w #%d,r%d\n", (int)((v >> 16) & 0xFFFF), dst, (int)(v & 0xFFFF), dst + 1);
		return;
	}
	case INDIR: {
		/* the address node: in x.kids, or a register variable read
		   (INDIR(VREGP), not listed in x.kids) */
		Node a = cse(p->kids[0]), ar = isaddrop4(a) ? NULL : p->x.kids[0] ? p->x.kids[0] : a;
		dst = rnum(p);
		if (ar && rnum(ar) == dst)	/* the address register is the high half */
			print("mov.w %s,r%d\nmov.w %s,r%d\n", ea4(a, ar, 2), dst + 1, ea4(a, ar, 0), dst);
		else
			print("mov.w %s,r%d\nmov.w %s,r%d\n", ea4(a, ar, 0), dst, ea4(a, ar, 2), dst + 1);
		return;
	}
	case ASGN: {
		Node a = cse(p->kids[0]), ar = NULL, val = p->x.kids[0];
		if (!isaddrop4(a)) {
			if (p->x.kids[1]) {	/* address and value both computed */
				ar = p->x.kids[0];
				val = p->x.kids[1];
			} else			/* address in a register variable */
				ar = a;
		}
		src = rnum(val);
		print("mov.w r%d,%s\nmov.w r%d,%s\n", src, ea4(a, ar, 0), src + 1, ea4(a, ar, 2));
		return;
	}
	case ARG:
		if (op == ARG+B) {
			int n = p->syms[0]->u.c.v.i;
			print("sub.w #%d,sp\nmov.w sp,r2\nmov.w #%d,r1\njsr @__blkcpy\n", roundup(n, 2), n);
			return;
		}
		src = rnum(p->x.kids[0]);
		print("mov.w r%d,@-sp\nmov.w r%d,@-sp\n", src + 1, src);
		return;
	case ADD: case SUB: case BAND: case BOR: case BXOR: {
		static char *lo[] = { "add.w", "sub.w", "and.w", "or.w", "xor.w" };
		static char *hi[] = { "addx.w", "subx.w", "and.w", "or.w", "xor.w" };
		int k = generic(op) == ADD ? 0 : generic(op) == SUB ? 1 : generic(op) == BAND ? 2
			: generic(op) == BOR ? 3 : 4;
		int b = rnum(p->x.kids[1]);
		dst = rnum(p);
		src = rnum(p->x.kids[0]);
		if (dst == b && dst != src) {	/* dst = a op dst */
			if (k == 1) {		/* a - dst = -dst + a */
				print("not.w r%d\nnot.w r%d\nadd.w #1,r%d\naddx.w #0,r%d\n", dst, dst + 1, dst + 1, dst);
				k = 0;
			}
			b = src;
		} else
			mov4(dst, src);
		print("%s r%d,r%d\n%s r%d,r%d\n", lo[k], b + 1, dst + 1, hi[k], b, dst);
		return;
	}
	case BCOM:
		dst = rnum(p);
		mov4(dst, rnum(p->x.kids[0]));
		print("not.w r%d\nnot.w r%d\n", dst, dst + 1);
		return;
	case NEG:
		dst = rnum(p);
		mov4(dst, rnum(p->x.kids[0]));
		print("not.w r%d\nnot.w r%d\nadd.w #1,r%d\naddx.w #0,r%d\n", dst, dst + 1, dst + 1, dst);
		return;
	case EQ: case NE: case GE: case GT: case LE: case LT: {
		char *lab = p->syms[0]->x.name;
		int a = rnum(p->x.kids[0]), b = rnum(p->x.kids[1]);
		int skip = genlabel(1);
		if (optype(op) == F) {
			print("jsr @__cmpf\ncmp.w #0,r0\n%s %s\n", bcc(op, 1), lab);
			return;
		}
		print("cmp.w r%d,r%d\n", b, a);
		switch (generic(op)) {
		case EQ:
			print("bne L_%s_c%d\ncmp.w r%d,r%d\nbeq %s\nL_%s_c%d:\n", tag, skip, b + 1, a + 1, lab, tag, skip);
			break;
		case NE:
			print("bne %s\ncmp.w r%d,r%d\nbne %s\n", lab, b + 1, a + 1, lab);
			break;
		default: {
			/* high words decide unless equal; then the low words, unsigned */
			char *strict = generic(op) == GE || generic(op) == GT
				? (optype(op) == I ? "bgt" : "bhi") : (optype(op) == I ? "blt" : "bcs");
			print("%s %s\nbne L_%s_c%d\ncmp.w r%d,r%d\n%s %s\nL_%s_c%d:\n",
				strict, lab, tag, skip, b + 1, a + 1, bcc(op, 0), lab, tag, skip);
		}
		}
		return;
	}
	}
	/* RET and register reads / writes: no code */
}

static void doarg(Node p) {
	assert(p && p->syms[0]);
	mkactual(2, p->syms[0]->u.c.v.i);
}

static void local(Symbol p) {
	if (askregvar(p, rmap(ttob(p->type))) == 0)
		mkauto(p);
}

static void function(Symbol f, Symbol caller[], Symbol callee[], int ncalls) {
	int i, saved, framesize;
	char list[40];

	usedmask[0] = usedmask[1] = 0;
	freemask[0] = freemask[1] = ~(unsigned)0;
	offset = 4;			/* saved FP, return address */
	for (i = 0; callee[i]; i++) {
		Symbol p = callee[i];
		Symbol q = caller[i];
		assert(q);
		offset = roundup(offset, 2);
		/* a char argument is pushed as a word: its byte is the low one */
		p->x.offset = q->x.offset = offset + (q->type->size == 1 ? 1 : 0);
		p->x.name = q->x.name = stringd(p->x.offset);
		p->sclass = q->sclass = AUTO;
		offset += roundup(q->type->size, 2);
	}
	assert(caller[i] == 0);
	offset = maxoffset = 0;
	gencode(caller, callee);
	framesize = roundup(maxoffset, 2);
	saved = usedmask[IREG] & vmask[IREG];
	list[0] = 0;
	for (i = 0; i < 8; i++)
		if (saved & (1 << i))
			sprintf(list + strlen(list), "%sr%d", list[0] ? "," : "", i);
	print("%s:\n", f->x.name);
	print("link fp,#%d\n", -framesize);
	if (saved)
		print("stm (%s),@-sp\n", list);
	emitcode();
	if (saved)
		print("ldm @sp+,(%s)\n", list);
	print("unlk fp\nrts\n");
}

static void defsymbol(Symbol p) {
	if (p->scope >= LOCAL && p->sclass == STATIC)
		p->x.name = stringf("S_%s_%d", tag, genlabel(1));
	else if (p->generated)
		p->x.name = stringf("L_%s_%s", tag, p->name);
	else if (p->scope == GLOBAL && p->sclass == STATIC)
		p->x.name = stringf("S_%s_%s", tag, p->name);
	else if (p->scope == GLOBAL || p->sclass == EXTERN)
		p->x.name = stringf("_%s", p->name);
	else if (p->scope == CONSTANTS && (isint(p->type) || isptr(p->type)))
		p->x.name = p->name;
	else
		p->x.name = p->name;
}

static void address(Symbol q, Symbol p, long n) {
	if (p->scope == GLOBAL || p->sclass == STATIC || p->sclass == EXTERN)
		q->x.name = stringf("%s%s%D", p->x.name, n >= 0 ? "+" : "", n);
	else {
		q->x.offset = p->x.offset + n;
		q->x.name = stringd(q->x.offset);
	}
}

/* the linker (h8link.py) splits the output on these markers */
static void segment(int n) {
	if (n == cseg)
		return;
	cseg = n;
	if (cseg == CODE)
		print(";@code\n");
	else if (cseg == BSS)
		print(";@bss\n");
	else if (cseg == DATA || cseg == LIT)
		print(";@data\n");
}

static void defconst(int suffix, int size, Value v) {
	if (suffix == F && size == 4) {
		float f = v.d;
		print("dc.w %d,%d\n", (int)((*(unsigned *)&f >> 16) & 0xFFFF), (int)(*(unsigned *)&f & 0xFFFF));
	} else if (size == 1)
		print("dc.b %d\n", (int)((suffix == I ? v.i : v.u) & 0xFF));
	else if (size == 2)
		print("dc.w %d\n", (int)((suffix == I ? v.i : suffix == P ? (unsigned long)v.p : v.u) & 0xFFFF));
	else if (size == 4) {
		unsigned long u = suffix == I ? (unsigned long)v.i : v.u;
		print("dc.w %d,%d\n", (int)((u >> 16) & 0xFFFF), (int)(u & 0xFFFF));
	} else
		assert(0);
}

static void defaddress(Symbol p) {
	print("dc.w (%s)&$FFFF\n", p->x.name);
}

static void defstring(int n, char *str) {
	int i;

	for (i = 0; i < n; i++)
		print("%s%d%s", i % 16 == 0 ? "dc.b " : "", str[i] & 0377,
			i % 16 == 15 || i == n - 1 ? "\n" : ",");
}

static void export(Symbol p) {}
static void import(Symbol p) {}

static void global(Symbol p) {
	if (p->type->align > 1)
		print("align 2\n");
	print("%s:\n", p->x.name);
	if (p->u.seg == BSS)
		print("ds.b %d\n", p->type->size);
}

static void space(int n) {
	if (cseg == BSS)
		return;
	while (n > 0) {
		int k = n > 16 ? 16 : n, i;
		print("dc.b 0");
		for (i = 1; i < k; i++)
			print(",0");
		print("\n");
		n -= k;
	}
}
